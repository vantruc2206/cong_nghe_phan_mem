\subsection{Hướng Dẫn Cài Đặt}

\begin{enumerate}
  \item \textbf{Clone repository:}
\begin{verbatim}
git clone https://github.com/[username]/[repository].git
cd [repository]
\end{verbatim}
  \item \textbf{Cài đặt phụ thuộc:}
  \begin{itemize}
    \item Frontend:
\begin{verbatim}
cd frontend
npm install
\end{verbatim}
    \item Backend:
\begin{verbatim}
cd backend
pip install -r requirements.txt
\end{verbatim}
  \end{itemize}
\end{enumerate}
